	public static function init() {
		add_action( 'admin_menu', array( __CLASS__, 'meniu' ), 20 );
		add_filter( 'petshop_ataskaitu_aprasai', function ( $a ) { $a[ self::PUSLAPIS ] = 'Kohortos, LTV pagal kontribuciją, pakartotiniai pirkimai, top klientai, rizikos sąrašas į Sender, RFM ir vieno kliento istorija.'; return $a; } );

	}

		add_submenu_page( self::TEVAS, 'Klientų analizė', 'Klientų analizė', 'manage_woocommerce', self::PUSLAPIS, array( __CLASS__, 'puslapis' ) );

	/* ---------- pagalbos ---------- */
	private static function u() { global $wpdb; return class_exists( 'Petshop_Ist_Adapteris' ) ? Petshop_Ist_Adapteris::lentele( 'u' ) : $wpdb->prefix . 'ps_fakt_uzsakymai'; }
	private static function e() { global $wpdb; return class_exists( 'Petshop_Ist_Adapteris' ) ? Petshop_Ist_Adapteris::lentele( 'e' ) : $wpdb->prefix . 'ps_fakt_eilutes'; }
	private static function d() { global $wpdb; return $wpdb->prefix . 'ps_dim_klientai'; }
	private static function yra( $t ) { global $wpdb; return $wpdb->get_var( $wpdb->prepare( 'SHOW TABLES LIKE %s', $t ) ) === $t; }
	public static function rodom_testinius() { return isset( $_GET['testiniai'] ) && '1' === $_GET['testiniai']; }
	private static function ts( $p = 'u' ) { return self::rodom_testinius() ? '' : " AND {$p}.testinis = 0 "; }
	private static function pradzia() { return class_exists( 'Petshop_Ataskaita_Pardavimai' ) ? Petshop_Ataskaita_Pardavimai::stat_pradzia() : (string) get_option( 'ps_stat_pradzia', '' ); }
	private static function nuo_apk( $nuo ) { $p = self::pradzia(); return ( $p && $nuo < $p ) ? $p : $nuo; }
	private static function fw_uzs( $a ) { return class_exists( 'Petshop_Analize_Filtrai' ) ? Petshop_Analize_Filtrai::uzs_atributai( $a ) : ''; }
	private static function fw_dim( $kol ) { return class_exists( 'Petshop_Analize_Filtrai' ) ? Petshop_Analize_Filtrai::dim_where( $kol ) : ''; }
	private static function eur( $ct ) { return Petshop_Ataskaitu_UI::eur( (int) $ct ); }
	private static function pr( $v, $sk = 1 ) { return null === $v ? '—' : number_format( $v, $sk, ',', ' ' ) . ' %'; }
	private static function url( $args = array() ) {
		$a = array_merge( array( 'page' => self::PUSLAPIS ), $args );
		if ( self::rodom_testinius() ) { $a['testiniai'] = '1'; }
		foreach ( array( 'f_nuo', 'f_iki', 'f_brendas', 'f_gyvunas', 'f_kat' ) as $fk ) { if ( isset( $_GET[ $fk ] ) && '' !== $_GET[ $fk ] && ! isset( $a[ $fk ] ) ) { $a[ $fk ] = sanitize_text_field( wp_unslash( $_GET[ $fk ] ) ); } }
		foreach ( array( 'preset', 'nuo', 'iki', 'saltinis' ) as $k ) { if ( isset( $_GET[ $k ] ) && ! isset( $a[ $k ] ) ) { $a[ $k ] = sanitize_text_field( wp_unslash( $_GET[ $k ] ) ); } }
		return admin_url( 'admin.php?' . http_build_query( $a ) );
	}
	private static function mediana( $arr ) {
		if ( ! $arr ) { return null; }
		sort( $arr ); $n = count( $arr );
		return $n % 2 ? $arr[ (int) floor( $n / 2 ) ] : ( $arr[ $n / 2 - 1 ] + $arr[ $n / 2 ] ) / 2;
	}
	private static function kliento_vardas( $hash, $kid ) {
		if ( $kid ) { $u = get_userdata( (int) $kid ); if ( $u ) { return trim( $u->first_name . ' ' . $u->last_name ) ?: $u->display_name; } }
		return 'svečias ' . substr( $hash, 0, 8 );
	}

	/** Visi užsakymai nuo pradžios: hash, nr, apmoketa_at, pajamos, kontr, kanalas. Vienas SELECT — toliau PHP. */
	private static function uzsakymai_visi() {
		global $wpdb; static $k = null;
		if ( null !== $k ) { return $k; }
		$u = self::u(); $t = self::ts(); $p = self::pradzia();
		$w = $p ? $wpdb->prepare( ' AND u.diena >= %s ', $p ) : '';
		if ( class_exists( 'Petshop_Analize_Filtrai' ) ) { $w .= Petshop_Analize_Filtrai::uzs_where( 'u' ); }
		$k = $wpdb->get_results( "SELECT u.uzsakymas_id, u.klientas_email_hash h, u.klientas_id, u.klientas_uzsakymo_nr nr, u.apmoketa_at, u.diena,
			u.prekiu_suma_ct - u.nuolaidu_ct paj, u.kontribucija_ct kontr, u.kanalas_pirmas kan, u.dienos_nuo_ankstesnio dna
			FROM $u u WHERE u.klientas_email_hash <> '' $t $w ORDER BY u.klientas_email_hash, u.apmoketa_at", ARRAY_A );
		return $k;
	}

	/** Per klientą: pirmas ts, kohorta, uzs, paj, kontr, kanalas, dienos iki 2-o, klientas_id. */
	private static function klientai() {
		static $c = null; if ( null !== $c ) { return $c; }
		$c = array();
		foreach ( self::uzsakymai_visi() as $o ) {
			$h = $o['h'];
			if ( ! isset( $c[ $h ] ) ) { $c[ $h ] = array( 'pirmas' => strtotime( $o['apmoketa_at'] . ' UTC' ), 'kohorta' => substr( $o['diena'], 0, 7 ), 'uzs' => 0, 'paj' => 0, 'kontr' => 0, 'kan' => (string) $o['kan'], 'iki2' => null, 'iki3' => null, 'kid' => (int) $o['klientas_id'], 'pask' => null, 'uzs_id' => array() ); }
			$c[ $h ]['uzs']++; $c[ $h ]['paj'] += (int) $o['paj']; $c[ $h ]['kontr'] += (int) $o['kontr'];
			$ts = strtotime( $o['apmoketa_at'] . ' UTC' ); $c[ $h ]['pask'] = $ts; $c[ $h ]['uzs_id'][] = (int) $o['uzsakymas_id'];
			if ( 2 === $c[ $h ]['uzs'] ) { $c[ $h ]['iki2'] = ( $ts - $c[ $h ]['pirmas'] ) / DAY_IN_SECONDS; }
			if ( 3 === $c[ $h ]['uzs'] ) { $c[ $h ]['iki3'] = ( $ts - $c[ $h ]['pirmas'] ) / DAY_IN_SECONDS; }
			if ( (int) $o['klientas_id'] ) { $c[ $h ]['kid'] = (int) $o['klientas_id']; }
		}
		return $c;
	}

	/** Apžvalgos KPI už laikotarpį (užsakymų lygiu) + visos istorijos rodikliai. */
	public static function apzvalga( $nuo, $iki ) {
		global $wpdb; $u = self::u(); $t = self::ts() . self::fw_uzs( 'u' );
		$r = $wpdb->get_row( $wpdb->prepare( "SELECT COUNT(*) n, SUM(u.klientas_naujas) nauji, COUNT(DISTINCT u.klientas_email_hash) klientu,
			SUM(u.prekiu_suma_ct - u.nuolaidu_ct) paj, SUM(CASE WHEN u.klientas_naujas=0 THEN u.prekiu_suma_ct - u.nuolaidu_ct ELSE 0 END) paj_grizt,
			SUM(u.kontribucija_ct) kontr, SUM(CASE WHEN u.klientas_naujas=0 THEN u.kontribucija_ct ELSE 0 END) kontr_grizt
			FROM $u u WHERE u.diena BETWEEN %s AND %s $t", $nuo, $iki ), ARRAY_A );
		$k = self::klientai(); $n = count( $k ); $pak = 0; $iki2 = array(); $iki3 = array(); $tri = 0; $dvi = 0; $leg = 0;
		foreach ( $k as $x ) { if ( $x['uzs'] >= 2 ) { $pak++; $dvi++; $iki2[] = $x['iki2']; } if ( $x['uzs'] >= 3 ) { $tri++; $iki3[] = $x['iki3']; } }
		if ( self::yra( self::d() ) ) { $leg = (int) $wpdb->get_var( 'SELECT COUNT(*) FROM ' . self::d() . ' WHERE legacy=1' . ( self::rodom_testinius() ? '' : ' AND testinis=0' ) ); }
		return array( 'lt' => $r, 'klientu_viso' => $n, 'pakartotinis_p' => $n ? $pak / $n * 100 : null, 'iki2_med' => self::mediana( $iki2 ), 'iki3_med' => self::mediana( $iki3 ),
			'dvi_i_tris_p' => $dvi ? $tri / $dvi * 100 : null, 'legacy' => $leg );
	}

	/** Kohortos: kohorta => [n, men => [klientu, kontr]] ; men = pilnų mėnesių nuo pirmo pirkimo (0..). */
	public static function kohortos() {
		$k = self::klientai(); $ko = array();
		foreach ( self::uzsakymai_visi() as $o ) {
			$h = $o['h']; $c = $k[ $h ]; $men = (int) floor( ( strtotime( $o['apmoketa_at'] . ' UTC' ) - $c['pirmas'] ) / ( 30.4375 * DAY_IN_SECONDS ) );
			if ( ! isset( $ko[ $c['kohorta'] ] ) ) { $ko[ $c['kohorta'] ] = array( 'n' => 0, 'men' => array(), 'hash' => array() ); }
			if ( ! isset( $ko[ $c['kohorta'] ]['hash'][ $h ] ) ) { $ko[ $c['kohorta'] ]['hash'][ $h ] = 1; $ko[ $c['kohorta'] ]['n']++; }
			if ( ! isset( $ko[ $c['kohorta'] ]['men'][ $men ] ) ) { $ko[ $c['kohorta'] ]['men'][ $men ] = array( 'kl' => array(), 'kontr' => 0 ); }
			$ko[ $c['kohorta'] ]['men'][ $men ]['kl'][ $h ] = 1; $ko[ $c['kohorta'] ]['men'][ $men ]['kontr'] += (int) $o['kontr'];
		}
		ksort( $ko ); return $ko;
	}

	/** LTV (kontribucija/klientą) per 90/180/365 d. pagal kohortą ir pirmą kanalą. Tik klientai, kurių langas jau pasibaigęs, kad nemeluotų. */
	public static function ltv( $pagal = 'kohorta' ) {
		$k = self::klientai(); $out = array(); $now = time();
		$ltv = array();
		foreach ( self::uzsakymai_visi() as $o ) {
			$h = $o['h']; $d = ( strtotime( $o['apmoketa_at'] . ' UTC' ) - $k[ $h ]['pirmas'] ) / DAY_IN_SECONDS;
			if ( ! isset( $ltv[ $h ] ) ) { $ltv[ $h ] = array( 90 => 0, 180 => 0, 365 => 0 ); }
			foreach ( array( 90, 180, 365 ) as $L ) { if ( $d <= $L ) { $ltv[ $h ][ $L ] += (int) $o['kontr']; } }
		}
		foreach ( $k as $h => $c ) {
			$g = 'kohorta' === $pagal ? $c['kohorta'] : ( $c['kan'] !== '' ? $c['kan'] : '(nežinomas)' );
			if ( ! isset( $out[ $g ] ) ) { $out[ $g ] = array( 'n' => 0, 90 => array( 0, 0 ), 180 => array( 0, 0 ), 365 => array( 0, 0 ) ); }
			$out[ $g ]['n']++;
			foreach ( array( 90, 180, 365 ) as $L ) { if ( ( $now - $c['pirmas'] ) / DAY_IN_SECONDS >= $L ) { $out[ $g ][ $L ][0] += $ltv[ $h ][ $L ]; $out[ $g ][ $L ][1]++; } }
		}
		ksort( $out ); return $out;
	}

	/** Rizikoje: refill laukiama_at praėjo ≥ N d., nepirko po jo. */
	public static function rizikoje( $riba_d = 7, $limit = 500 ) {
		global $wpdb; if ( ! self::yra( self::d() ) ) { return array(); }
		$d = self::d(); $u = self::u(); $w = ( self::rodom_testinius() ? '' : ' AND k.testinis=0 ' ) . self::fw_dim( 'k.raktas' );
		return $wpdb->get_results( $wpdb->prepare( "SELECT k.raktas, k.klientas_id, k.uzsakymu_sk, k.kontribucija_ct, k.refill_laukiama_at, k.paskutinis_pirkimas_at, k.pagrindinis_brendas, k.segmentas,
			COALESCE(
				(SELECT o.billing_email FROM {$wpdb->prefix}wc_orders o JOIN $u f ON f.uzsakymas_id = o.id WHERE f.klientas_email_hash = k.raktas ORDER BY f.apmoketa_at DESC LIMIT 1),
				(SELECT i.email FROM {$wpdb->prefix}ps_ist_fakt_uzsakymai f2 JOIN {$wpdb->prefix}ps_ist_uzsakymai i ON i.id = f2.uzsakymas_id - 900000000 WHERE f2.klientas_email_hash = k.raktas ORDER BY f2.apmoketa_at DESC LIMIT 1)
			) email
			FROM $d k WHERE k.refill_laukiama_at IS NOT NULL AND k.refill_laukiama_at < UTC_TIMESTAMP() - INTERVAL %d DAY
			  AND (k.paskutinis_pirkimas_at IS NULL OR k.paskutinis_pirkimas_at < k.refill_laukiama_at) $w
			ORDER BY k.kontribucija_ct DESC LIMIT %d", $riba_d, $limit ), ARRAY_A );
	}

	public static function rizikos_csv() {
		if ( ! current_user_can( 'manage_woocommerce' ) ) { wp_die( 'Neturite teisių.' ); }
		check_admin_referer( 'ps_klientai_csv' );
		if ( isset( $_GET['testiniai'] ) ) { $_GET['testiniai'] = '1'; }
		header( 'Content-Type: text/csv; charset=UTF-8' ); header( 'Content-Disposition: attachment; filename="petshop_rizikoje_' . gmdate( 'Ymd' ) . '.csv"' );
		$o = fopen( 'php://output', 'w' ); fputs( $o, "\xEF\xBB\xBF" );
		fputcsv( $o, array( 'email', 'vardas', 'uzsakymu', 'kontribucija_eur', 'refill_laukta', 'paskutinis_pirkimas', 'brendas', 'segmentas' ), ';' );
		foreach ( self::rizikoje( (int) get_option( 'ps_rytas_ribos' )['refill_d'] ?: 7, 5000 ) as $r ) {
			if ( empty( $r['email'] ) ) { continue; }
			fputcsv( $o, array( $r['email'], self::kliento_vardas( $r['raktas'], $r['klientas_id'] ), $r['uzsakymu_sk'], number_format( $r['kontribucija_ct'] / 100, 2, '.', '' ), substr( (string) $r['refill_laukiama_at'], 0, 10 ), substr( (string) $r['paskutinis_pirkimas_at'], 0, 10 ), $r['pagrindinis_brendas'], $r['segmentas'] ), ';' );
		}
		fclose( $o ); exit;
	}

	public static function puslapis() {
		if ( ! current_user_can( 'manage_woocommerce' ) ) { wp_die( 'Neturite teisių.' ); }
		$UI = 'Petshop_Ataskaitu_UI';
		$tab = isset( $_GET['tab'] ) ? sanitize_key( $_GET['tab'] ) : 'apzvalga';
		$tabai = array( 'apzvalga' => 'Apžvalga', 'kohortos' => 'Kohortos', 'ltv' => 'LTV', 'top' => 'Top klientai', 'rizika' => 'Rizikoje', 'rfm' => 'RFM' );
		if ( ! isset( $tabai[ $tab ] ) && 'klientas' !== $tab ) { $tab = 'apzvalga'; }
		$lt = $UI::laikotarpis(); $nuo = self::nuo_apk( $lt['nuo'] );
		$fper = class_exists( 'Petshop_Analize_Filtrai' ) && Petshop_Analize_Filtrai::periodinis();
		if ( $fper ) { list( $nuo, $lt['iki'] ) = Petshop_Analize_Filtrai::periodas( $nuo, $lt['iki'] ); $lt['nuo'] = $nuo; $nuo = self::nuo_apk( $nuo ); }

		$UI::antraste( 'Klientų analizė', 'Kas grįžta, kiek klientas atneša per gyvenimą ir ką reikia pasivyti. Pinigai — kontribucija, ne pajamos (eShoprent istorijoje kontribucijos nėra — S1546).', 'faktai' );
		if ( class_exists( 'Petshop_Ist_Adapteris' ) ) { echo Petshop_Ist_Adapteris::jungiklis_html(); }
		if ( class_exists( 'Petshop_Analize_Filtrai' ) ) { echo Petshop_Analize_Filtrai::juosta_html(); }
		if ( self::rodom_testinius() ) { echo '<div class="psru-spejimas"><b>Rodomi IR testiniai įrašai.</b> <a href="' . esc_url( remove_query_arg( 'testiniai' ) ) . '">✕ slėpti</a></div>'; }
		if ( 'klientas' === $tab ) { self::tab_klientas( $UI ); $UI::pabaiga(); return; }

		echo '<div class="psru-tabai">';
		foreach ( $tabai as $k => $v ) { echo '<a class="psru-tab' . ( $k === $tab ? ' akt' : '' ) . '" href="' . esc_url( self::url( array( 'tab' => $k ) ) ) . '">' . esc_html( $v ) . '</a>'; }
		echo '</div>';
		if ( ! self::uzsakymai_visi() ) { $UI::tuscia( 'Užsakymų nėra.', self::pradzia() ? 'Statistikos pradžia: ' . self::pradzia() : '' ); $UI::pabaiga(); return; }
		call_user_func( array( __CLASS__, 'tab_' . $tab ), $UI, $lt, $nuo );
		$UI::pabaiga();
	}

	private static function tab_apzvalga( $UI, $lt, $nuo ) {
		if ( ! ( class_exists( 'Petshop_Analize_Filtrai' ) && Petshop_Analize_Filtrai::periodinis() ) ) { $UI::juosta_v2( self::PUSLAPIS, $lt ); }
		$a = self::apzvalga( $nuo, $lt['iki'] ); $r = $a['lt'];
		$paj = (int) $r['paj']; $kontr = (int) $r['kontr'];
		$UI::kpi_juosta( array(
			array( 'antraste' => 'Pakartotinis pirkimas', 'reiksme' => self::pr( $a['pakartotinis_p'], 0 ), 'tooltip' => 'Klientai su ≥ 2 užsakymais / visi klientai nuo pradžios.' ),
			array( 'antraste' => 'Laikas iki 2-o pirkimo', 'reiksme' => null === $a['iki2_med'] ? '—' : number_format( $a['iki2_med'], 0, ',', ' ' ) . ' d.', 'tooltip' => 'Mediana. Nuo pirmo iki antro apmokėto užsakymo.' ),
			array( 'antraste' => 'Grįžtančių dalis kontribucijoje', 'reiksme' => self::pr( $kontr ? (int) $r['kontr_grizt'] / $kontr * 100 : null, 0 ), 'tooltip' => 'Laikotarpio kontribucija iš grįžtančių klientų / visa kontribucija.' ),
			array( 'antraste' => 'Klientai laikotarpyje', 'reiksme' => number_format( (int) $r['klientu'], 0, ',', ' ' ) . ' <span style="font-size:.5em;font-weight:400;color:#787c82">' . (int) $r['nauji'] . ' nauji</span>', 'tooltip' => 'Unikalūs pirkę klientai per laikotarpį.' ),
		), array(
			'Klientų iš viso' => number_format( $a['klientu_viso'], 0, ',', ' ' ),
			'2 → 3 pirkimas' => self::pr( $a['dvi_i_tris_p'], 0 ),
			'Iki 3-o (mediana)' => null === $a['iki3_med'] ? '—' : number_format( $a['iki3_med'], 0, ',', ' ' ) . ' d.',
			'Grįžtančių dalis pajamose' => self::pr( $paj ? (int) $r['paj_grizt'] / $paj * 100 : null, 0 ),
			'Legacy klientai' => number_format( $a['legacy'], 0, ',', ' ' ),
		) );
		/* S1546 #3: petshop.lt režime — augimo atskaitos taškas be maišymo. */
		if ( class_exists( 'Petshop_Ist_Adapteris' ) && 'wc' === Petshop_Ist_Adapteris::saltinis() && Petshop_Ist_Adapteris::paruostas() ) {
			global $wpdb; $iu = Petshop_Ist_Adapteris::iu();
			$pn = gmdate( 'Y-m-d', strtotime( $nuo . ' -1 year' ) ); $pi = gmdate( 'Y-m-d', strtotime( $lt['iki'] . ' -1 year' ) );
			$per = $wpdb->get_row( $wpdb->prepare( "SELECT COUNT(*) n, SUM(prekiu_suma_ct) paj, COUNT(DISTINCT klientas_email_hash) kl, SUM(klientas_naujas) nauji FROM $iu WHERE diena BETWEEN %s AND %s", $pn, $pi ), ARRAY_A );
			if ( $per && (int) $per['n'] ) {
				echo '<p class="psru-pastaba" style="margin-top:8px"><b>Tas pats laikotarpis eShoprent pernai</b> (' . esc_html( $pn ) . ' – ' . esc_html( $pi ) . '): ' . (int) $per['n'] . ' užs. · ' . esc_html( self::eur( (int) $per['paj'] ) ) . ' · ' . (int) $per['kl'] . ' klientai (' . (int) $per['nauji'] . ' nauji).</p>';
			}
		}
		/* segmentai */
		global $wpdb;
		if ( self::yra( self::d() ) ) {
			$seg = $wpdb->get_results( 'SELECT k.segmentas, COUNT(*) n, SUM(k.kontribucija_ct) k FROM ' . self::d() . ' k WHERE 1=1' . ( self::rodom_testinius() ? '' : ' AND k.testinis=0' ) . self::fw_dim( 'k.raktas' ) . ' GROUP BY k.segmentas ORDER BY n DESC', ARRAY_A );
			if ( $seg ) {
				echo '<h2 style="margin-top:20px">Segmentai (šiandien)</h2>';

	private static function tab_kohortos( $UI, $lt, $nuo ) {
		$ko = self::kohortos(); $maxm = 0;
		foreach ( $ko as $k ) { foreach ( array_keys( $k['men'] ) as $m ) { $maxm = max( $maxm, $m ); } }
		$maxm = min( $maxm, 12 );
		echo '<p class="psru-aprasas">Eilutė — pirmo pirkimo mėnuo. Stulpelis — kelintas mėnuo po pirmo pirkimo. Viršuje: kiek % kohortos klientų tą mėnesį pirko. Apačioje: kumuliatyvi kontribucija vienam kohortos klientui.</p>';
		echo '<div style="overflow:auto"><table class="psru-lent psk-koh"><thead><tr><th style="text-align:left">Kohorta</th><th>Klientų</th>';
		for ( $m = 0; $m <= $maxm; $m++ ) { echo '<th>M' . $m . '</th>'; }
		echo '</tr></thead><tbody>';
		foreach ( $ko as $kh => $k ) {
			echo '<tr><td style="text-align:left"><b>' . esc_html( $kh ) . '</b></td><td>' . (int) $k['n'] . '</td>';
			$kum = 0; $pilnu = (int) floor( ( time() - strtotime( $kh . '-01' ) ) / ( 30.4375 * DAY_IN_SECONDS ) );
			for ( $m = 0; $m <= $maxm; $m++ ) {
				if ( $m > $pilnu ) { echo '<td class="psk-tuscia"></td>'; continue; }
				$kl = isset( $k['men'][ $m ] ) ? count( $k['men'][ $m ]['kl'] ) : 0; $kum += isset( $k['men'][ $m ] ) ? $k['men'][ $m ]['kontr'] : 0;
				$p = $k['n'] ? $kl / $k['n'] * 100 : 0; $al = min( 0.85, 0.08 + $p / 100 * 0.9 );
				$al = ( 0 === $m ) ? 0.06 : $al;
				echo '<td style="background:rgba(19,94,150,' . $al . ');color:' . ( $al > 0.5 ? '#fff' : '#1d2327' ) . '"><b>' . number_format( $p, 0 ) . ' %</b><br><span style="font-size:11px;opacity:.85">' . esc_html( self::eur( $k['n'] ? $kum / $k['n'] : 0 ) ) . '</span></td>';
			}
			echo '</tr>';
		}
		echo '</tbody></table></div><style>.psk-koh td,.psk-koh th{text-align:center;padding:7px 8px;font-size:12.5px;border-bottom:1px solid #f0f0f1}.psk-koh .psk-tuscia{background:#fafafa}</style>';
		echo '<p class="psru-pastaba">M0 = pirmo pirkimo mėnuo (visada 100 %). Tušti langeliai — mėnuo dar nepraėjo. Mėnuo = 30,44 d. nuo pirmo pirkimo, ne kalendorinis.</p>';
	}
