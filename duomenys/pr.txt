	private static function e() { global $wpdb; return class_exists( 'Petshop_Ist_Adapteris' ) ? Petshop_Ist_Adapteris::lentele( 'e' ) : $wpdb->prefix . 'ps_fakt_eilutes'; }
	private static function k() { global $wpdb; return $wpdb->prefix . 'ps_fakt_kainos'; }
	private static function a() { global $wpdb; return $wpdb->prefix . 'ps_fakt_atsargos_d'; }
	private static function yra( $t ) { global $wpdb; return $wpdb->get_var( $wpdb->prepare( 'SHOW TABLES LIKE %s', $t ) ) === $t; }
	private static function eur( $ct ) { return Petshop_Ataskaitu_UI::eur( (int) $ct ); }
	private static function preke( $id ) { $p = get_the_title( (int) $id ); return '<a href="' . esc_url( get_edit_post_link( (int) $id ) ) . '">' . esc_html( mb_substr( html_entity_decode( $p ? $p : '#' . $id ), 0, 48 ) ) . '</a>'; }
	private static function url( $args ) { $a = array_merge( array( 'page' => self::PUSLAPIS ), $args ); if ( isset( $_GET['saltinis'] ) && ! isset( $a['saltinis'] ) ) { $a['saltinis'] = sanitize_key( $_GET['saltinis'] ); } foreach ( array( 'f_nuo', 'f_iki', 'f_brendas', 'f_gyvunas', 'f_kat' ) as $fk ) { if ( isset( $_GET[ $fk ] ) && '' !== $_GET[ $fk ] && ! isset( $a[ $fk ] ) ) { $a[ $fk ] = sanitize_text_field( wp_unslash( $_GET[ $fk ] ) ); } } return admin_url( 'admin.php?' . http_build_query( $a ) ); }
	private static function fw_eil( $a ) { return class_exists( 'Petshop_Analize_Filtrai' ) ? Petshop_Analize_Filtrai::eil_where( $a ) : ''; }
	private static function fw_tax( $pid ) { return class_exists( 'Petshop_Analize_Filtrai' ) ? Petshop_Analize_Filtrai::preke_tax_where( $pid ) : ''; }
	private static function pradzia() { return class_exists( 'Petshop_Ataskaita_Pardavimai' ) ? Petshop_Ataskaita_Pardavimai::stat_pradzia() : ''; }

	/** ABC×XYZ iš 365 d. eilučių. Grąžina [prekės => [...], matrica => [A][X] => [n, paj]]. */
	public static function abc_xyz( $dienu = 365 ) {
		global $wpdb; $e = self::e();
		$nuo = gmdate( 'Y-m-d', time() - $dienu * DAY_IN_SECONDS ); $iki = gmdate( 'Y-m-d' );
		if ( class_exists( 'Petshop_Analize_Filtrai' ) ) { list( $nuo, $iki ) = Petshop_Analize_Filtrai::periodas( $nuo, $iki ); }
		$p = self::pradzia(); if ( $p && $nuo < $p ) { $nuo = $p; }
		$fw = self::fw_eil( 'e' );
		$r = $wpdb->get_results( $wpdb->prepare( "SELECT e.preke_id, YEARWEEK(e.diena, 3) sav, SUM(e.kiekis) k, SUM(e.kaina_ct) paj, SUM(e.savikaina_ct) sav_ct FROM $e e WHERE e.preke_id > 0 AND e.diena BETWEEN %s AND %s AND e.testinis = 0 $fw GROUP BY e.preke_id, sav", $nuo, $iki ), ARRAY_A );
		$nesusieta = $wpdb->get_row( $wpdb->prepare( "SELECT COUNT(*) n, COALESCE(SUM(e.kaina_ct),0) paj FROM $e e WHERE e.preke_id = 0 AND e.diena BETWEEN %s AND %s AND e.testinis = 0", $nuo, $iki ), ARRAY_A );
		$sav_n = max( 1, (int) ceil( ( strtotime( $iki ) - strtotime( $nuo ) ) / WEEK_IN_SECONDS ) );
		$pr = array();
		foreach ( $r as $x ) { $id = (int) $x['preke_id']; if ( ! isset( $pr[ $id ] ) ) { $pr[ $id ] = array( 'paj' => 0, 'sav_ct' => 0, 'k' => 0, 'sav' => array() ); } $pr[ $id ]['paj'] += (int) $x['paj']; $pr[ $id ]['sav_ct'] += (int) $x['sav_ct']; $pr[ $id ]['k'] += (int) $x['k']; $pr[ $id ]['sav'][ $x['sav'] ] = (int) $x['k']; }
		uasort( $pr, function ( $a, $b ) { return $b['paj'] <=> $a['paj']; } );
		$viso = 0; foreach ( $pr as $x ) { $viso += $x['paj']; }
		$kum = 0; $mat = array();
		foreach ( $pr as $id => &$x ) {
			$kum += $x['paj']; $d = $viso ? $kum / $viso : 1;
			$x['abc'] = $d <= 0.80 ? 'A' : ( $d <= 0.95 ? 'B' : 'C' );
			$vals = array_values( $x['sav'] ); while ( count( $vals ) < $sav_n ) { $vals[] = 0; }
			$mu = array_sum( $vals ) / count( $vals ); $var = 0; foreach ( $vals as $v ) { $var += ( $v - $mu ) ** 2; } $cv = $mu > 0 ? sqrt( $var / count( $vals ) ) / $mu : 9;
			$x['cv'] = $cv; $x['xyz'] = $cv < 0.5 ? 'X' : ( $cv < 1.0 ? 'Y' : 'Z' );
			$x['dalis'] = $viso ? $x['paj'] / $viso * 100 : 0;
			if ( ! isset( $mat[ $x['abc'] ][ $x['xyz'] ] ) ) { $mat[ $x['abc'] ][ $x['xyz'] ] = array( 'n' => 0, 'paj' => 0 ); }
			$mat[ $x['abc'] ][ $x['xyz'] ]['n']++; $mat[ $x['abc'] ][ $x['xyz'] ]['paj'] += $x['paj'];
			unset( $x['sav'] );
		}
		return array( 'prekes' => $pr, 'matrica' => $mat, 'viso' => $viso, 'nuo' => $nuo, 'iki' => $iki, 'savaiciu' => $sav_n, 'nesusieta' => $nesusieta );
	}

	/** Neparduodama ≥ N d.: prekės su likučiu (AV) arba paskelbtos, be pardavimo N d. */
	public static function neparduodama( $d = 60 ) {
		global $wpdb; $e = self::e(); $a = self::a();
		$snap = (string) $wpdb->get_var( "SELECT MAX(data) FROM $a" ); if ( ! $snap ) { return array(); }
		return $wpdb->get_results( $wpdb->prepare(
			"SELECT x.preke_id, x.likutis_av, x.likutis_tiekejo, x.verte_ct, x.savikaina_vnt_ct,
			        (SELECT MAX(e.diena) FROM $e e WHERE e.preke_id = x.preke_id AND e.testinis = 0) pask,
			        (SELECT MAX(k.diena) FROM " . self::k() . " k WHERE k.preke_id = x.preke_id) pask_kaina
			 FROM $a x WHERE x.data = %s AND x.testinis = 0 AND ( x.likutis_av > 0 OR x.likutis_tiekejo > 0 ) " . self::fw_tax( 'x.preke_id' ) . "
			 HAVING pask IS NULL OR pask < %s ORDER BY x.verte_ct DESC LIMIT 300", $snap, gmdate( 'Y-m-d', time() - $d * DAY_IN_SECONDS ) ), ARRAY_A );
	}

	/** Kainų pokyčiai su pardavimais 14 d. prieš/po. */
	public static function kainos( $limit = 60 ) {
		global $wpdb; $k = self::k(); $e = self::e(); if ( ! self::yra( $k ) ) { return array(); }
		$r = $wpdb->get_results( $wpdb->prepare( "SELECT k.id, k.preke_id, k.diena, k.kaina_ct, k.reguliari_ct, k.akcijoje, k.saltinis,
			(SELECT k2.kaina_ct FROM $k k2 WHERE k2.preke_id = k.preke_id AND k2.nuo_at < k.nuo_at ORDER BY k2.nuo_at DESC LIMIT 1) pries_ct
			FROM $k k WHERE k.testinis = 0 " . self::fw_tax( 'k.preke_id' ) . ( class_exists( 'Petshop_Analize_Filtrai' ) ? Petshop_Analize_Filtrai::eil_periodas( 'k' ) : '' ) . " ORDER BY k.nuo_at DESC LIMIT %d", $limit ), ARRAY_A );
		foreach ( $r as &$x ) {
			$x['po']    = (float) $wpdb->get_var( $wpdb->prepare( "SELECT COALESCE(SUM(kiekis),0) FROM $e WHERE preke_id = %d AND diena >= %s AND diena < DATE_ADD(%s, INTERVAL 14 DAY) AND testinis = 0", $x['preke_id'], $x['diena'], $x['diena'] ) );
			$x['pries'] = (float) $wpdb->get_var( $wpdb->prepare( "SELECT COALESCE(SUM(kiekis),0) FROM $e WHERE preke_id = %d AND diena < %s AND diena >= DATE_SUB(%s, INTERVAL 14 DAY) AND testinis = 0", $x['preke_id'], $x['diena'], $x['diena'] ) );
			$x['dienu_po'] = min( 14, max( 1, (int) floor( ( time() - strtotime( $x['diena'] ) ) / DAY_IN_SECONDS ) ) );
		}
		return $r;
	}

	/** Akcijų uplift: kiekviena akcija — vnt./d. ir kontribucija/d. akcijos metu vs 28 d. prieš. */
	public static function akcijos() {
		global $wpdb; $A = $wpdb->prefix . 'ps_akcijos'; $AP = $wpdb->prefix . 'ps_akciju_prekes'; $e = self::e();
		if ( ! self::yra( $A ) || ! self::yra( $AP ) ) { return array(); }
		$ak = $wpdb->get_results( "SELECT a.id, a.pavadinimas, a.busena, a.metodas, a.reiksme, a.nuo, a.iki, (SELECT COUNT(*) FROM $AP p WHERE p.akcija_id = a.id) prekiu FROM $A a WHERE a.busena IN ('aktyvi','pasibaigusi','archyvas') AND a.nuo IS NOT NULL AND a.iki IS NOT NULL ORDER BY a.nuo DESC LIMIT 40", ARRAY_A );
		if ( class_exists( 'Petshop_Analize_Filtrai' ) && Petshop_Analize_Filtrai::periodinis() ) {
			list( $fn, $fi ) = Petshop_Analize_Filtrai::periodas( '0000-01-01', '9999-12-31' );
			$ak = array_values( array_filter( $ak, function ( $x ) use ( $fn, $fi ) { return substr( $x['iki'], 0, 10 ) >= $fn && substr( $x['nuo'], 0, 10 ) <= $fi; } ) );
		}
		global $wpdb;
		foreach ( $ak as &$x ) {
			$nuo = substr( $x['nuo'], 0, 10 ); $iki = min( substr( $x['iki'], 0, 10 ), gmdate( 'Y-m-d' ) );
			$d_ak = max( 1, (int) floor( ( strtotime( $iki ) - strtotime( $nuo ) ) / DAY_IN_SECONDS ) + 1 );
			$b_nuo = gmdate( 'Y-m-d', strtotime( $nuo ) - 28 * DAY_IN_SECONDS );
			$q = function ( $n, $i ) use ( $wpdb, $e, $AP, $x ) { return $wpdb->get_row( $wpdb->prepare( "SELECT COALESCE(SUM(e.kiekis),0) k, COALESCE(SUM(e.kaina_ct),0) paj, COALESCE(SUM(e.kaina_ct - COALESCE(e.savikaina_ct,0)),0) ant FROM $e e WHERE e.preke_id IN (SELECT product_id FROM $AP WHERE akcija_id = %d) AND e.diena BETWEEN %s AND %s AND e.testinis = 0 " . self::fw_eil( 'e' ) . "", $x['id'], $n, $i ), ARRAY_A ); };
			$m = $q( $nuo, $iki ); $b = $q( $b_nuo, gmdate( 'Y-m-d', strtotime( $nuo ) - DAY_IN_SECONDS ) );
			$x['d_ak'] = $d_ak; $x['k_d'] = $m['k'] / $d_ak; $x['b_k_d'] = $b['k'] / 28; $x['ant_d'] = $m['ant'] / $d_ak; $x['b_ant_d'] = $b['ant'] / 28; $x['paj'] = (int) $m['paj']; $x['ant'] = (int) $m['ant'];
			$x['uplift'] = $x['b_k_d'] > 0 ? ( $x['k_d'] / $x['b_k_d'] - 1 ) * 100 : null;
		}
		return $ak;
	}

	public static function puslapis() {
		if ( ! current_user_can( 'manage_woocommerce' ) ) { wp_die( 'Neturite teisių.' ); }
		$UI = 'Petshop_Ataskaitu_UI';
		$tab = isset( $_GET['tab'] ) ? sanitize_key( $_GET['tab'] ) : 'abc';
		$tabai = array( 'abc' => 'ABC × XYZ', 'nepard' => 'Neparduodama', 'kainos' => 'Kainų pokyčiai', 'akcijos' => 'Akcijų uplift' );
		if ( ! isset( $tabai[ $tab ] ) ) { $tab = 'abc'; }
		$UI::antraste( 'Prekių analizė', 'Kurios prekės neša pinigus, kurios guli, ką padarė kainų pokyčiai ir ar akcijos uždirbo (ne tik pardavė).', 'faktai' );
		if ( class_exists( 'Petshop_Ist_Adapteris' ) ) { echo Petshop_Ist_Adapteris::jungiklis_html(); }
		if ( class_exists( 'Petshop_Analize_Filtrai' ) ) { echo Petshop_Analize_Filtrai::juosta_html(); }
		echo '<div class="psru-tabai">'; foreach ( $tabai as $k => $v ) { echo '<a class="psru-tab' . ( $k === $tab ? ' akt' : '' ) . '" href="' . esc_url( self::url( array( 'tab' => $k ) ) ) . '">' . esc_html( $v ) . '</a>'; } echo '</div>';
		call_user_func( array( __CLASS__, 'tab_' . $tab ), $UI );
		$UI::pabaiga();
	}

	private static function tab_abc( $UI ) {
		$d = self::abc_xyz(); $m = $d['matrica'];
		if ( ! $d['prekes'] ) { $UI::tuscia( 'Pardavimų eilučių nėra.' ); return; }
		echo '<p class="psru-aprasas">Pajamos ' . esc_html( $d['nuo'] . ' – ' . $d['iki'] ) . ' (' . (int) $d['savaiciu'] . ' sav.). A/B/C — 80/95 % pajamų. X/Y/Z — savaitinės paklausos stabilumas. AX — laikyti visada; CZ — tik dropship.</p>';
		echo '<table class="psru-lent psp-m"><thead><tr><th></th><th>X stabili</th><th>Y svyruoja</th><th>Z nereguliari</th><th>Iš viso</th></tr></thead><tbody>';
		foreach ( array( 'A', 'B', 'C' ) as $r ) { echo '<tr><th>' . $r . '</th>'; $rn = 0; $rp = 0;
			foreach ( array( 'X', 'Y', 'Z' ) as $c ) { $x = isset( $m[ $r ][ $c ] ) ? $m[ $r ][ $c ] : array( 'n' => 0, 'paj' => 0 ); $rn += $x['n']; $rp += $x['paj'];
				echo '<td><a href="' . esc_url( self::url( array( 'tab' => 'abc', 'kl' => $r . $c ) ) ) . '"><b>' . (int) $x['n'] . '</b> prekių</a><br><span class="psru-mut">' . esc_html( self::eur( $x['paj'] ) ) . ' · ' . number_format( $d['viso'] ? $x['paj'] / $d['viso'] * 100 : 0, 0 ) . ' %</span></td>'; }
			echo '<td><b>' . $rn . '</b><br><span class="psru-mut">' . esc_html( self::eur( $rp ) ) . '</span></td></tr>'; }
		echo '</tbody></table><style>.psp-m td,.psp-m th{text-align:center;padding:10px 12px;border-bottom:1px solid #f0f0f1}.psp-m td a{text-decoration:none}</style>';
		if ( ! empty( $d['nesusieta'] ) && (int) $d['nesusieta']['n'] ) { echo '<p class="psru-pastaba"><b>Nebeparduodamos</b> (eShoprent eilutės be WC atitikmens): ' . (int) $d['nesusieta']['n'] . ' eil. · ' . esc_html( self::eur( (int) $d['nesusieta']['paj'] ) ) . ' — į matricą neįtraukta (S1546 #4).</p>'; }
		$kl = isset( $_GET['kl'] ) ? strtoupper( sanitize_key( $_GET['kl'] ) ) : '';
		$rows = array(); $i = 0;
		foreach ( $d['prekes'] as $id => $x ) { if ( $kl && $x['abc'] . $x['xyz'] !== $kl ) { continue; } if ( ++$i > 300 ) { break; }
			$rows[] = array( self::preke( $id ), '<b>' . $x['abc'] . $x['xyz'] . '</b>', (int) $x['k'], self::eur( $x['paj'] ), number_format( $x['dalis'], 1, ',', ' ' ) . ' %', $x['sav_ct'] ? self::eur( $x['paj'] - $x['sav_ct'] ) : '—', $x['sav_ct'] ? number_format( ( $x['paj'] - $x['sav_ct'] ) / $x['sav_ct'] * 100, 1, ',', ' ' ) . ' %' : '—', number_format( $x['cv'], 2, ',', ' ' ) ); }
		echo '<h2>' . ( $kl ? 'Klasė ' . esc_html( $kl ) : 'Visos prekės' ) . ( $kl ? ' · <a href="' . esc_url( self::url( array( 'tab' => 'abc' ) ) ) . '">visos</a>' : '' ) . '</h2>';
		$UI::lentele( 'psp-abc', array( array( 'pav' => 'Prekė', 'kaire' => true ), array( 'pav' => 'Klasė' ), array( 'pav' => 'Vnt.' ), array( 'pav' => 'Pajamos' ), array( 'pav' => 'Dalis' ), array( 'pav' => 'Antkainis €' ), array( 'pav' => 'Antkainis %' ), array( 'pav' => 'CV' ) ), $rows );
	}

	private static function tab_akcijos( $UI ) {
		$r = self::akcijos();
		echo '<p class="psru-aprasas">Akcijos prekių pardavimai per dieną akcijos metu vs 28 d. prieš. Svarbiausia — antkainis/d.: pardavei daugiau, bet ar uždirbai daugiau? Tik petshop.lt (S1546 #4).</p>';

		$rows = array();
