 * Plugin Name: Petshop Ataskaita Atsargos
 * Description: Atsargos ir pirkimas (planas v2.0 §4.4) — ką užsakyti, stockout €, apyvartumas, sell-through, galiojimas, dropship prieinamumas.
 * Version: 1.2
 *
 * ŠALTINIAI. `ps_fakt_atsargos_d` (naktinis snapshot: likutis_av, pardavimai
 * 7/30 d., dienų atsargų, stockout, vertė) · `ps_partijos` (gavimai, likę,
 * galiojimas) · `ps_sources` (tiekėjas, tiekėjo likutis) · `ps_fakt_eilutes`
 * (COGS). Ekranas NIEKO neužsako — siūlo kiekį, sprendžia žmogus (§2 rule).
 *
 * AV vs DROPSHIP: snapshot'e `likutis_av IS NULL` = prekė ne AV lentynoje.
 * Pirkimo siūlymai — tik AV. Dropship'ui — atskiras skirtukas apie tiekėjo
 * prieinamumą (kiek dienų per 30 tiekėjas neturėjo).
 *
 * SIŪLOMAS KIEKIS = paklausa per (lead + buferis) − likutis − rezervuota +
 * atsarga 30 d. Paklausa = `pardavimai_vnt_30d / 30`. Be duomenų (30 d. = 0)
 * — nesiūlom nieko: geriau tyla nei išgalvotas skaičius.
 */

class Petshop_Ataskaita_Atsargos {
	const VERSIJA  = '1.2';
	const TEVAS    = 'petshop-reports';
	const PUSLAPIS = 'ps-atsargos';

	public static function init() {
		add_action( 'admin_menu', array( __CLASS__, 'meniu' ), 20 );
		add_filter( 'petshop_ataskaitu_aprasai', function ( $a ) { $a[ self::PUSLAPIS ] = 'Ką užsakyti į AV sandėlį, kiek kainuoja tuščios lentynos, apyvartumas, gavimų sell-through, galiojimai ir dropship tiekėjų prieinamumas.'; return $a; } );
	}
	public static function meniu() { add_submenu_page( self::TEVAS, 'Atsargos ir pirkimas', 'Atsargos ir pirkimas', 'manage_woocommerce', self::PUSLAPIS, array( __CLASS__, 'puslapis' ) ); }

	private static function a() { global $wpdb; return $wpdb->prefix . 'ps_fakt_atsargos_d'; }
	private static function e() { global $wpdb; return class_exists( 'Petshop_Ist_Adapteris' ) ? Petshop_Ist_Adapteris::lentele( 'e' ) : $wpdb->prefix . 'ps_fakt_eilutes'; }
	private static function pt() { global $wpdb; return $wpdb->prefix . 'ps_partijos'; }
	private static function src() { global $wpdb; return $wpdb->prefix . 'ps_sources'; }
	private static function yra( $t ) { global $wpdb; return $wpdb->get_var( $wpdb->prepare( 'SHOW TABLES LIKE %s', $t ) ) === $t; }
	private static function fw_tax( $pid ) { return class_exists( 'Petshop_Analize_Filtrai' ) ? Petshop_Analize_Filtrai::preke_tax_where( $pid ) : ''; }
	private static function eur( $ct ) { return Petshop_Ataskaitu_UI::eur( (int) $ct ); }
	private static function riba( $k, $d ) { $r = get_option( 'ps_rytas_ribos', array() ); return ( is_array( $r ) && isset( $r[ $k ] ) && $r[ $k ] !== '' ) ? (float) $r[ $k ] : $d; }
	private static function tiek_vardas( $k ) { return class_exists( 'Petshop_Ataskaita_Pardavimai' ) ? Petshop_Ataskaita_Pardavimai::tiekejo_vardas( $k ) : $k; }
	private static function preke( $id ) { $p = get_the_title( (int) $id ); return '<a href="' . esc_url( get_edit_post_link( (int) $id ) ) . '">' . esc_html( mb_substr( html_entity_decode( $p ? $p : '#' . $id ), 0, 48 ) ) . '</a>'; }
	private static function url( $args ) { $a = array_merge( array( 'page' => self::PUSLAPIS ), $args ); if ( isset( $_GET['saltinis'] ) && ! isset( $a['saltinis'] ) ) { $a['saltinis'] = sanitize_key( $_GET['saltinis'] ); } foreach ( array( 'f_nuo', 'f_iki', 'f_brendas', 'f_gyvunas', 'f_kat' ) as $fk ) { if ( isset( $_GET[ $fk ] ) && '' !== $_GET[ $fk ] && ! isset( $a[ $fk ] ) ) { $a[ $fk ] = sanitize_text_field( wp_unslash( $_GET[ $fk ] ) ); } } return admin_url( 'admin.php?' . http_build_query( $a ) ); }
	private static function snap() { global $wpdb; static $d = null; if ( null === $d ) { $d = (string) $wpdb->get_var( 'SELECT MAX(data) FROM ' . self::a() ); } return $d; }

	/** Ką užsakyti — AV prekės, kurių atsargų dienų < lead. */
	public static function uzsakyti( $lead ) {
		global $wpdb; $a = self::a(); $s = self::src(); $snap = self::snap(); if ( ! $snap ) { return array(); }
		$r = $wpdb->get_results( $wpdb->prepare(
			"SELECT x.preke_id, x.likutis_av, x.rezervuota, x.pardavimai_vnt_30d p30, x.pardavimai_vnt_7d p7, x.dienu_atsargu, x.savikaina_vnt_ct sav, x.artimiausia_galiojimas,
			        (SELECT sr.source FROM $s sr WHERE sr.product_id = x.preke_id AND sr.is_active = 1 ORDER BY sr.priority LIMIT 1) tiekejas
			 FROM $a x WHERE x.data = %s AND x.likutis_av IS NOT NULL AND x.pardavimai_vnt_30d > 0 AND x.testinis = 0 " . self::fw_tax( 'x.preke_id' ) . "
			   AND (x.likutis_av - x.rezervuota) / (x.pardavimai_vnt_30d / 30) < %f
			 ORDER BY x.pardavimai_vnt_30d * x.savikaina_vnt_ct DESC LIMIT 300", $snap, $lead ), ARRAY_A );
		foreach ( $r as &$x ) {
			$rate = $x['p30'] / 30; $laisva = max( 0, (int) $x['likutis_av'] - (int) $x['rezervuota'] );
			$x['dienu'] = $rate > 0 ? $laisva / $rate : null;
			$x['siulyti'] = max( 0, (int) ceil( $rate * ( $lead + 30 ) - $laisva ) );
			$x['suma_ct'] = $x['siulyti'] * (int) $x['sav'];
		}
		return $r;
	}

	/** Stockout laikotarpyje: dienos be likučio × dienos paklausa × kaina = prarastos pajamos (konservatyviai). */
	public static function stockout( $nuo, $iki ) {
		global $wpdb; $a = self::a();
		if ( class_exists( 'Petshop_Analize_Filtrai' ) ) { list( $nuo, $iki ) = Petshop_Analize_Filtrai::periodas( $nuo, $iki ); }
		return $wpdb->get_results( $wpdb->prepare(
			"SELECT x.preke_id, SUM(x.stockout) dienos, COUNT(*) snapu, AVG(x.pardavimai_vnt_30d) / 30 rate, MAX(x.likutis_av IS NOT NULL) av,
			        (SELECT pm.meta_value FROM {$wpdb->postmeta} pm WHERE pm.post_id = x.preke_id AND pm.meta_key = '_price' LIMIT 1) kaina,
			        (SELECT pm2.meta_value FROM {$wpdb->postmeta} pm2 WHERE pm2.post_id = x.preke_id AND pm2.meta_key = '_ps_abc' LIMIT 1) abc
			 FROM $a x WHERE x.data BETWEEN %s AND %s AND x.testinis = 0 " . self::fw_tax( 'x.preke_id' ) . " GROUP BY x.preke_id HAVING dienos > 0 AND rate > 0
			 ORDER BY dienos * rate * kaina DESC LIMIT 100", $nuo, $iki ), ARRAY_A );
	}

	/** Apyvartumas: atsargų vertė pagal dieną + COGS 365 d. */
	public static function apyvartumas() {
		global $wpdb; $a = self::a(); $e = self::e();
		$ist = $wpdb->get_results( "SELECT data, SUM(verte_ct) verte, SUM(likutis_av IS NOT NULL AND likutis_av > 0) su_likuciu, SUM(stockout) stockout, COUNT(*) prekiu FROM $a WHERE testinis = 0 GROUP BY data ORDER BY data DESC LIMIT 90", ARRAY_A );
		$cogs = (int) $wpdb->get_var( $wpdb->prepare( "SELECT COALESCE(SUM(e.savikaina_ct),0) FROM $e e WHERE e.diena >= %s AND e.testinis = 0", gmdate( 'Y-m-d', time() - 365 * DAY_IN_SECONDS ) ) );
		$vid = 0; if ( $ist ) { $s = 0; foreach ( $ist as $x ) { $s += (int) $x['verte']; } $vid = $s / count( $ist ); }
		$snap = self::snap(); $buck = array();
		if ( $snap ) { $buck = $wpdb->get_results( $wpdb->prepare( "SELECT CASE WHEN dienu_atsargu IS NULL THEN 'be paklausos' WHEN dienu_atsargu < 7 THEN '< 7 d.' WHEN dienu_atsargu < 30 THEN '7–30 d.' WHEN dienu_atsargu < 90 THEN '30–90 d.' ELSE '> 90 d.' END b, COUNT(*) n, SUM(verte_ct) v FROM $a WHERE data = %s AND likutis_av > 0 AND testinis = 0 GROUP BY b", $snap ), ARRAY_A ); }
		return array( 'istorija' => array_reverse( $ist ), 'cogs365' => $cogs, 'vid_verte' => $vid, 'apyvarta' => $vid > 0 ? $cogs / $vid : null, 'buck' => $buck );
	}

	/** Sell-through per gavimą (partiją). */
	public static function sell_through() {
		global $wpdb; $pt = self::pt(); if ( ! self::yra( $pt ) ) { return array(); }
		return $wpdb->get_results( "SELECT id, product_id, gauta, kiekis_gautas, kiekis_liko, savikaina_eur, tiekejas, geriausia_iki, DATEDIFF(CURDATE(), gauta) dienu FROM $pt WHERE atsaukta = 0 AND kiekis_gautas > 0 " . self::fw_tax( 'product_id' ) . " ORDER BY gauta DESC LIMIT 200", ARRAY_A );
	}

	public static function galiojimas( $d ) {
		global $wpdb; $pt = self::pt(); if ( ! self::yra( $pt ) ) { return array(); }
		return $wpdb->get_results( $wpdb->prepare( "SELECT product_id, geriausia_iki, SUM(kiekis_liko) liko, SUM(kiekis_liko * savikaina_eur * 100) verte_ct, DATEDIFF(geriausia_iki, CURDATE()) liko_d FROM $pt WHERE atsaukta = 0 AND kiekis_liko > 0 " . self::fw_tax( 'product_id' ) . " AND geriausia_iki IS NOT NULL AND geriausia_iki <= DATE_ADD(CURDATE(), INTERVAL %d DAY) GROUP BY product_id, geriausia_iki ORDER BY geriausia_iki", $d ), ARRAY_A );
	}

	/** Dropship: per tiekėją — prekių, be likučio dabar, stockout dienų dalis per 30 d. */
	public static function dropship() {
		global $wpdb; $s = self::src(); $a = self::a();
		$dabar = $wpdb->get_results( "SELECT source, COUNT(*) n, SUM(stock_qty <= 0) be_likucio, SUM(is_sellable = 0) neparduodama FROM $s WHERE is_active = 1 GROUP BY source", ARRAY_A );
		$so = $wpdb->get_results( $wpdb->prepare( "SELECT sr.source, SUM(x.likutis_tiekejo = 0) so_dienos, COUNT(*) snapu FROM $a x JOIN $s sr ON sr.product_id = x.preke_id AND sr.is_active = 1 WHERE x.data >= %s AND x.likutis_tiekejo IS NOT NULL AND x.testinis = 0 GROUP BY sr.source", gmdate( 'Y-m-d', time() - 30 * DAY_IN_SECONDS ) ), ARRAY_A );
		$m = array(); foreach ( $so as $x ) { $m[ $x['source'] ] = $x; }
		foreach ( $dabar as &$d ) { $d['so_dalis'] = isset( $m[ $d['source'] ] ) && $m[ $d['source'] ]['snapu'] ? $m[ $d['source'] ]['so_dienos'] / $m[ $d['source'] ]['snapu'] * 100 : null; }
		return $dabar;
	}

	public static function puslapis() {
		if ( ! current_user_can( 'manage_woocommerce' ) ) { wp_die( 'Neturite teisių.' ); }
		$UI = 'Petshop_Ataskaitu_UI';
		$tab = isset( $_GET['tab'] ) ? sanitize_key( $_GET['tab'] ) : 'uzsakyti';
		$tabai = array( 'uzsakyti' => 'Ką užsakyti', 'stockout' => 'Stockout €', 'apyvarta' => 'Apyvartumas', 'sell' => 'Sell-through', 'galiojimas' => 'Galiojimas', 'dropship' => 'Dropship' );
		if ( ! isset( $tabai[ $tab ] ) ) { $tab = 'uzsakyti'; }
		$UI::antraste( 'Atsargos ir pirkimas', 'AV sandėlis: ką užsakyti, kiek kainuoja tuščios lentynos, kaip sukasi pinigai. Sistema siūlo — sprendi tu.', 'faktai' );
		if ( class_exists( 'Petshop_Ist_Adapteris' ) ) { echo Petshop_Ist_Adapteris::jungiklis_html(); echo '<p class="psru-mut" style="margin-top:-8px">Atsargų istorija ir savikainos — tik petshop.lt (S1546 #4); eShoprent čia prisideda tik prie paklausos pjūvių.</p>'; }
		echo '<div class="psru-tabai">'; foreach ( $tabai as $k => $v ) { echo '<a class="psru-tab' . ( $k === $tab ? ' akt' : '' ) . '" href="' . esc_url( self::url( array( 'tab' => $k ) ) ) . '">' . esc_html( $v ) . '</a>'; } echo '</div>';
		if ( ! self::snap() ) { $UI::tuscia( 'Atsargų snapshot dar nėra.', 'Naktinis cron `ps_fakt_atsargos_d` dar nesuveikė.' ); $UI::pabaiga(); return; }
		echo '<p class="psru-mut" style="margin-top:10px">Snapshot: ' . esc_html( self::snap() ) . '</p>';
		call_user_func( array( __CLASS__, 'tab_' . $tab ), $UI );
		$UI::pabaiga();
	}

	private static function tab_uzsakyti( $UI ) {
		$lead = self::riba( 'lead_d', 7 ); $r = self::uzsakyti( $lead );
		echo '<p class="psru-aprasas">AV prekės, kurių laisvo likučio užteks mažiau nei ' . (int) $lead . ' d. (tiekėjo lead + buferis, keičiama Rytas → Ribos). Siūlomas kiekis = paklausa per lead + 30 d. − laisvas likutis.</p>';
		if ( ! $r ) { $UI::tuscia( 'Užsakyti nieko nereikia.', 'Visų AV prekių su paklausa likučio užtenka ilgiau nei ' . (int) $lead . ' d.' ); return; }
		$rows = array(); $suma = 0; $pt = array();
		foreach ( $r as $x ) { $suma += $x['suma_ct']; $t = (string) $x['tiekejas']; $pt[ $t ] = ( isset( $pt[ $t ] ) ? $pt[ $t ] : 0 ) + $x['suma_ct'];
			$rows[] = array( self::preke( $x['preke_id'] ), esc_html( self::tiek_vardas( $t ) ), (int) $x['likutis_av'], (int) $x['rezervuota'], (int) $x['p30'], (int) $x['p7'], null === $x['dienu'] ? '—' : number_format( $x['dienu'], 1, ',', ' ' ), '<b>' . (int) $x['siulyti'] . '</b>', self::eur( $x['sav'] ), self::eur( $x['suma_ct'] ), $x['artimiausia_galiojimas'] ? esc_html( $x['artimiausia_galiojimas'] ) : '' ); }
		$UI::lentele( 'psa-uz', array( array( 'pav' => 'Prekė', 'kaire' => true ), array( 'pav' => 'Tiekėjas' ), array( 'pav' => 'Likutis' ), array( 'pav' => 'Rezerv.' ), array( 'pav' => 'Pard. 30 d.' ), array( 'pav' => '7 d.' ), array( 'pav' => 'Užteks d.' ), array( 'pav' => 'Užsakyti vnt.' ), array( 'pav' => 'Savikaina' ), array( 'pav' => 'Suma' ), array( 'pav' => 'Galiojimas' ) ), $rows );
		echo '<p><b>Iš viso užsakymui: ' . esc_html( self::eur( $suma ) ) . '</b> — ';
		$d = array(); foreach ( $pt as $t => $v ) { $d[] = esc_html( self::tiek_vardas( $t ) ) . ' ' . esc_html( self::eur( $v ) ); } echo implode( ' · ', $d ) . '</p>';
	}

	private static function tab_stockout( $UI ) {
		$lt = $UI::laikotarpis(); $UI::juosta_v2( self::PUSLAPIS, $lt );
		$r = self::stockout( $lt['nuo'], $lt['iki'] );
		echo '<p class="psru-aprasas">Prarastos pajamos = dienos be likučio × vidutinė dienos paklausa × kaina. Konservatyvu: dalis pirkėjų būtų nusipirkę kitą prekę.</p>';
		if ( ! $r ) { $UI::tuscia( 'Stockout su paklausa laikotarpyje nebuvo.' ); return; }
		$rows = array(); $viso = 0;
		foreach ( $r as $x ) { $pr = (int) round( $x['dienos'] * $x['rate'] * (float) $x['kaina'] * 100 ); $viso += $pr;
			$rows[] = array( self::preke( $x['preke_id'] ), esc_html( (string) $x['abc'] ), $x['av'] ? 'AV' : 'dropship', (int) $x['dienos'] . ' / ' . (int) $x['snapu'], number_format( $x['rate'], 2, ',', ' ' ), number_format( (float) $x['kaina'], 2, ',', ' ' ) . ' €', '<b>' . self::eur( $pr ) . '</b>' ); }
		$UI::lentele( 'psa-so', array( array( 'pav' => 'Prekė', 'kaire' => true ), array( 'pav' => 'ABC' ), array( 'pav' => 'Sandėlis' ), array( 'pav' => 'Dienų be likučio' ), array( 'pav' => 'Paklausa vnt./d.' ), array( 'pav' => 'Kaina' ), array( 'pav' => 'Prarasta' ) ), $rows );
		echo '<p><b>Iš viso prarasta (įvertis): ' . esc_html( self::eur( $viso ) ) . '</b></p>';
	}

	private static function tab_apyvarta( $UI ) {
		$a = self::apyvartumas();
		$UI::kpi_juosta( array(
			array( 'antraste' => 'Atsargų vertė (AV)', 'reiksme' => self::eur( $a['istorija'] ? end( $a['istorija'] )['verte'] : 0 ), 'tooltip' => 'Paskutinis snapshot, savikaina be PVM.' ),
			array( 'antraste' => 'Apyvartumas / metus', 'reiksme' => null === $a['apyvarta'] ? '—' : number_format( $a['apyvarta'], 1, ',', ' ' ) . '×', 'tooltip' => 'COGS 365 d. / vidutinė atsargų vertė.' ),
			array( 'antraste' => 'COGS 365 d.', 'reiksme' => self::eur( $a['cogs365'] ) ),
			array( 'antraste' => 'Vid. vertė', 'reiksme' => self::eur( $a['vid_verte'] ), 'tooltip' => 'Paskutinių 90 snapshot\'ų vidurkis.' ),
		) );
		if ( $a['buck'] ) { $rows = array(); foreach ( $a['buck'] as $b ) { $rows[] = array( esc_html( $b['b'] ), (int) $b['n'], self::eur( $b['v'] ) ); }
			echo '<h2>Kiek dienų užteks (AV prekės su likučiu)</h2>'; $UI::lentele( 'psa-b', array( array( 'pav' => 'Atsargų dienų', 'kaire' => true ), array( 'pav' => 'Prekių' ), array( 'pav' => 'Vertė' ) ), $rows, array( 'paieska' => false, 'csv' => false ) );
			echo '<p class="psru-pastaba">„> 90 d." — užšaldyti pinigai: kandidatai akcijai arba neužsakinėti. „be paklausos" — likutis yra, per 30 d. neparduota nė vieno.</p>'; }
		if ( count( $a['istorija'] ) > 1 ) { $rows = array(); foreach ( array_slice( array_reverse( $a['istorija'] ), 0, 30 ) as $x ) { $rows[] = array( esc_html( $x['data'] ), self::eur( $x['verte'] ), (int) $x['su_likuciu'], (int) $x['stockout'] ); }
			echo '<h2>Istorija (30 paskutinių snapshot\'ų)</h2>'; $UI::lentele( 'psa-i', array( array( 'pav' => 'Data', 'kaire' => true ), array( 'pav' => 'Vertė' ), array( 'pav' => 'AV prekių su likučiu' ), array( 'pav' => 'Stockout' ) ), $rows, array( 'paieska' => false ) ); }
	}

	private static function tab_sell( $UI ) {
		$r = self::sell_through();
		echo '<p class="psru-aprasas">Kiekvienas gavimas (partija): kiek % parduota per laiką nuo gavimo. Lėtos partijos — kandidatai akcijai; greitos — užsakyti daugiau.</p>';
		if ( ! $r ) { $UI::tuscia( 'Gavimų (partijų) dar nėra.' ); return; }
		$rows = array();
		foreach ( $r as $x ) { $parduota = (int) $x['kiekis_gautas'] - (int) $x['kiekis_liko']; $p = $x['kiekis_gautas'] ? $parduota / $x['kiekis_gautas'] * 100 : 0; $d = max( 1, (int) $x['dienu'] );
			$rows[] = array( self::preke( $x['product_id'] ), esc_html( $x['gauta'] ), esc_html( self::tiek_vardas( (string) $x['tiekejas'] ) ), (int) $x['kiekis_gautas'], $parduota, (int) $x['kiekis_liko'], $d, '<b>' . number_format( $p, 0 ) . ' %</b>', number_format( $parduota / $d * 30, 1, ',', ' ' ), $x['kiekis_liko'] > 0 && $parduota > 0 ? number_format( $x['kiekis_liko'] / ( $parduota / $d ), 0, ',', ' ' ) . ' d.' : '—', self::eur( $x['kiekis_liko'] * $x['savikaina_eur'] * 100 ), $x['geriausia_iki'] ? esc_html( $x['geriausia_iki'] ) : '' ); }
		$UI::lentele( 'psa-st', array( array( 'pav' => 'Prekė', 'kaire' => true ), array( 'pav' => 'Gauta' ), array( 'pav' => 'Tiekėjas' ), array( 'pav' => 'Vnt.' ), array( 'pav' => 'Parduota' ), array( 'pav' => 'Liko' ), array( 'pav' => 'Dienų' ), array( 'pav' => 'Sell-through' ), array( 'pav' => 'Tempas /30 d.' ), array( 'pav' => 'Liks parduoti' ), array( 'pav' => 'Likučio vertė' ), array( 'pav' => 'Galioja iki' ) ), $rows );
	}

		foreach ( array( 30, 60, 90, 180 ) as $x ) { echo ( $x === $d ? '<b>' . $x . ' d.</b>' : '<a href="' . esc_url( self::url( array( 'tab' => 'galiojimas', 'd' => $x ) ) ) . '">' . $x . ' d.</a>' ) . ' '; }

		echo '<p><b>Rizikoje: ' . esc_html( self::eur( $v ) ) . '</b> savikainos. <a href="' . esc_url( admin_url( 'admin.php?page=petshop-akcijos' ) ) . '">Kurti akciją →</a></p>';
	}

	private static function tab_dropship( $UI ) {
		$r = self::dropship();
		echo '<p class="psru-aprasas">Dropship tiekėjai: kiek prekių šiuo metu be likučio ir kokią dienų dalį per 30 d. tiekėjas neturėjo prekės (iš naktinių snapshot\'ų).</p>';
		if ( ! $r ) { $UI::tuscia( 'ps_sources tuščia.' ); return; }
		$rows = array();
		foreach ( $r as $x ) { $rows[] = array( esc_html( self::tiek_vardas( $x['source'] ) ), (int) $x['n'], (int) $x['be_likucio'], number_format( $x['n'] ? $x['be_likucio'] / $x['n'] * 100 : 0, 0 ) . ' %', (int) $x['neparduodama'], null === $x['so_dalis'] ? '—' : number_format( $x['so_dalis'], 1, ',', ' ' ) . ' %' ); }
		$UI::lentele( 'psa-d', array( array( 'pav' => 'Tiekėjas', 'kaire' => true ), array( 'pav' => 'Prekių' ), array( 'pav' => 'Be likučio dabar' ), array( 'pav' => 'Dalis' ), array( 'pav' => 'Neparduodamos' ), array( 'pav' => 'Stockout dienų dalis 30 d.' ) ), $rows, array( 'paieska' => false ) );
		echo '<p class="psru-pastaba">Aukšta „stockout dienų dalis" = tiekėjas nepatikimas šioms prekėms: verta imti į AV arba ieškoti alternatyvos.</p>';
	}
